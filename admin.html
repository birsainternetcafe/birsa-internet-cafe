const db = firebase.firestore(); const orderList = document.getElementById("orderList"); db.collection("orders") .orderBy("time", "desc") .onSnapshot((snapshot) => { orderList.innerHTML = ""; // clear old list snapshot.forEach((doc) => { let order = doc.data(); let li = document.createElement("li"); li.textContent = order.name + " - ₹" + order.amount + " (" + order.service + ")"; orderList.appendChild(li); }); });
